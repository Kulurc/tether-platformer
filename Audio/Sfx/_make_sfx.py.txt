"""Procedural SFX for Tether. Everything is synthesized here (no samples).
Writes 44.1 kHz mono 16-bit WAVs into the folder given as argv[1]."""
import sys, os, wave
import numpy as np
from scipy import signal

SR = 44100
OUT = sys.argv[1]
os.makedirs(OUT, exist_ok=True)


def t_axis(dur):
    return np.arange(int(dur * SR)) / SR


def noise(dur, rng):
    return rng.standard_normal(int(dur * SR))


def bp(x, lo, hi, order=2):
    sos = signal.butter(order, [lo, hi], btype="bandpass", fs=SR, output="sos")
    return signal.sosfilt(sos, x)


def lp(x, f, order=2):
    return signal.sosfilt(signal.butter(order, f, btype="lowpass", fs=SR, output="sos"), x)


def hp(x, f, order=2):
    return signal.sosfilt(signal.butter(order, f, btype="highpass", fs=SR, output="sos"), x)


def env(n, a, d, shape=1.0):
    """attack (s), then exponential-ish decay over d seconds"""
    t = np.arange(n) / SR
    e = np.where(t < a, t / max(a, 1e-6), np.exp(-(t - a) / max(d, 1e-6) * 4.0) ** shape)
    return e


def sweep_bp(x, f0, f1, q=2.0, steps=48):
    """band-pass whose centre glides from f0 to f1 (block-wise, crossfaded)"""
    out = np.zeros_like(x)
    n = len(x)
    blk = max(n // steps, 64)
    win = np.hanning(blk * 2)
    for i in range(0, n, blk):
        c = f0 * (f1 / f0) ** (i / n)
        lo, hi = c / (1 + 1 / q), c * (1 + 1 / q)
        seg = x[max(0, i - blk // 2): i + blk + blk // 2]
        y = bp(seg, max(lo, 30), min(hi, SR / 2 - 100))
        a = i - max(0, i - blk // 2)
        piece = y[a: a + blk]
        out[i: i + len(piece)] += piece
    return out


def reverb(x, size=0.6, mix=0.18, damp=4000, seed=7):
    rng = np.random.default_rng(seed)
    n = int(size * SR)
    ir = rng.standard_normal(n) * np.exp(-np.arange(n) / SR / size * 5.0)
    ir = lp(ir, damp)
    ir[: int(0.012 * SR)] = 0
    ir /= np.sqrt(np.sum(ir ** 2)) + 1e-9
    wet = signal.fftconvolve(x, ir)
    wet = np.concatenate([wet, np.zeros(max(0, len(x) + n - len(wet)))])[: len(x) + n]
    dry = np.concatenate([x, np.zeros(n)])
    return dry * (1 - mix) + wet * mix * 1.6


def fade(x, fin=0.002, fout=0.02):
    x = x.copy()
    a, b = int(fin * SR), int(fout * SR)
    if a: x[:a] *= np.linspace(0, 1, a)
    if b: x[-b:] *= np.linspace(1, 0, b)
    return x


def trim(x, thresh=1e-4):
    idx = np.where(np.abs(x) > thresh * np.max(np.abs(x)))[0]
    return x[: idx[-1] + 1] if len(idx) else x


def save(name, x, peak_db=-1.0, loop=False):
    x = np.asarray(x, dtype=np.float64)
    if not loop:
        x = fade(trim(x), 0.001, 0.03)
    x = x / (np.max(np.abs(x)) + 1e-12) * 10 ** (peak_db / 20)
    data = (np.clip(x, -1, 1) * 32767).astype(np.int16)
    with wave.open(os.path.join(OUT, name + ".wav"), "wb") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(SR)
        w.writeframes(data.tobytes())
    print(f"{name:22s} {len(x)/SR:5.2f}s")


def thump(dur, f0, f1, decay, rng=None):
    t = t_axis(dur)
    f = f0 * (f1 / f0) ** (t / dur)
    ph = 2 * np.pi * np.cumsum(f) / SR
    return np.sin(ph) * env(len(t), 0.003, decay)


def grains(dur, count, lo, hi, gdur, rng, decay_over=None, amp_jitter=0.6):
    """sparse little noise grains (grass blades, gravel, crackle)"""
    x = np.zeros(int(dur * SR))
    for _ in range(count):
        t0 = rng.random() ** (1.6 if decay_over else 1.0) * (decay_over or dur)
        g = int(gdur * SR * rng.uniform(0.5, 1.5))
        i = int(t0 * SR)
        if i + g >= len(x): continue
        b = bp(rng.standard_normal(g), lo * rng.uniform(0.8, 1.2), hi * rng.uniform(0.85, 1.1))
        a = (1 - amp_jitter + amp_jitter * rng.random())
        if decay_over: a *= np.exp(-t0 / decay_over * 2.5)
        x[i: i + g] += b * np.hanning(g) * a
    return x

for k in range(4):  # grass footsteps: a soft crunch of blades + a muffled heel

    heel = lp(noise(d, rng), 600) * env(int(d * SR), 0.002, 0.03) * 0.5 + thump(d, 140, 90, 0.035) * 0.35
    save(f"step_grass_{k+1}", reverb(lp(rust, 6500) + heel * 1.3, 0.25, 0.08), -3)

rng = np.random.default_rng(9)

def make_loop(x, xfade):
    """seamless loop: crossfade the tail into the head"""
    n = int(xfade * SR)
    head, body, tail = x[:n], x[n:-n], x[-n:]
    w = np.linspace(0, 1, n)
    return np.concatenate([body, tail * (1 - w) + head * w])


rng = np.random.default_rng(12)
d = 2.6  # sever strain (loop): rope fibres stretching and creaking under load
t = t_axis(d)
rate = 55 + 25 * np.sin(2 * np.pi * 0.7 * t)
ph = np.cumsum(rate) / SR
pulses = (np.diff(np.floor(ph), prepend=0) > 0).astype(float) * (0.5 + 0.5 * rng.random(len(t)))
creak = bp(pulses, 260, 380) * 1.0 + bp(pulses, 800, 1100) * 0.5
groan = lp(signal.sawtooth(2 * np.pi * np.cumsum(62 + 3 * np.sin(2 * np.pi * 0.9 * t)) / SR), 300) * 0.25
fibres = hp(grains(d, 180, 2500, 7000, 0.003, rng), 2000) * 0.35
x = creak + groan + fibres
save("sever_strain_loop", make_loop(x, 0.3), -4, loop=True)

rng = np.random.default_rng(13)
bpm = 118.0  # phase III heartbeat (loop): fast, heavy, close - with a low uneasy drone
beat = 60.0 / bpm
n = int(round(4 * beat * SR))
t = np.arange(n) / SR
x = np.zeros(n + int(0.4 * SR))  # tails past the end wrap to the start
for b in range(4):
    for off, f0, a, dc in [(0.0, 62, 1.0, 0.11), (0.26, 74, 0.75, 0.09)]:
        i = int((b * beat + off) * SR)
        m = int(0.35 * SR)
        tt = np.arange(m) / SR
        s = np.sin(2 * np.pi * np.cumsum(f0 * (0.75 + 0.25 * np.exp(-tt * 20))) / SR) * env(m, 0.004, dc)
        s += lp(rng.standard_normal(m), 160) * env(m, 0.003, dc * 0.6) * 0.35
        seg = s
        x[i: i + len(seg)] += a * seg
x[: len(x) - n] += x[n:]
x = x[:n]
loop_len = n / SR
drone_f = round(41 * loop_len) / loop_len           # whole cycles per loop: seamless
drone = (np.sin(2 * np.pi * drone_f * t) + 0.5 * np.sin(2 * np.pi * drone_f * 1.5 * t + 1)) * 0.12
drone *= 0.8 + 0.2 * np.sin(2 * np.pi * t / loop_len)
whine_f = round(5200 * loop_len) / loop_len
whine = np.sin(2 * np.pi * whine_f * t) * 0.006 * (0.6 + 0.4 * np.sin(2 * np.pi * 2 * t / loop_len))
y = np.tanh((x + drone) * 1.8) + whine
save("heartbeat_loop", y, -2, loop=True)
